# A drum bar copies every sample, including simultaneous hits.
wait 2000
tap SEQ
spin PRESET 4
tapkey 0
tapkey 1
spin ALGO 2
tapkey 5
expect seq_events 3
press EDIT
tap REC
spin ALGO 1
tap SEL
release EDIT
expect seq_events 6
spin SELECT -5
expect seq_note 0
spin SELECT 1
expect seq_note 1
spin ALGO 2
spin SELECT 4
expect seq_note 5
press EDIT
spin ALGO 1
release EDIT
tapkey 11
expect seq_events 7
press EDIT
tap SEL
release EDIT
expect seq_events 7
tap SEL
expect seq_events 9
spin SELECT 2
expect seq_note 0
spin SELECT 1
expect seq_note 1
# An empty copied bar can clear an occupied bar after confirmation.
press EDIT
spin ALGO 1
tap REC
spin ALGO -1
tap SEL
release EDIT
expect seq_events 9
tap SEL
expect seq_events 6
tap HOME
