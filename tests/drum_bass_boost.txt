# K4 boosts the LM-1 bus for both keyboard and sequencer drums.
wait 2000
expect solo5 32
turn SELECT 4
expect instrument 5
expect drum_bass_boost 0
wav drum_boost_off.wav
key 0 down 100
wait 500
key 0 up
wait 100
wavstop
turn K4 8
expect drum_bass_boost 8
wav drum_boost_on.wav
key 0 down 100
wait 500
key 0 up
wait 100
wavstop
turn K4 5
expect drum_bass_boost 8
turn SELECT 1
turn K4 -8
expect drum_bass_boost 8
turn SELECT -1
turn K4 -20
expect drum_bass_boost 0
tap PLAY
expect playback_mode 1
tap PLAY
expect seq_playing 1
wav drum_sequence_off.wav
wait 15000
expect hits5 >0
tap PLAY
wavstop
tap HOME
wait 500
turn K4 8
expect drum_bass_boost 8
tap PLAY
expect playback_mode 1
tap PLAY
expect seq_playing 1
wav drum_sequence_on.wav
wait 15000
expect hits5 >0
tap PLAY
wavstop
tap HOME
reboot
wait 2000
expect drum_bass_boost 0
