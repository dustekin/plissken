# SEQ tap runs the four-track drum sequencer; drum keys pressed while it runs are recorded on 16th steps.
wait 2000
turn SELECT 4
expect instrument 5
expect seq_run 0
expect seq_steps 0
key 0 down 100
wait 100
key 0 up
wait 100
expect seq_steps 0
tap SEQ
wait 200
expect seq_run 1
key 0 down 100
wait 100
key 0 up
wait 100
key 6 down 100
wait 100
key 6 up
wait 100
expect seq_steps >0
wait 3000
# SEQ held + key 2 clears the pattern
press SEQ
wait 500
key 2 down 100
wait 100
key 2 up
wait 100
release SEQ
wait 200
expect seq_steps 0
expect seq_run 1
tap SEQ
wait 200
expect seq_run 0
