# A long note may not be pasted beyond the 128-bar limit.
wait 2000
tap SEQ
master 4096
tapkey 0
expect seq_events 1
press EDIT
tap REC
spin ALGO 127
tap SEL

expect seq_warning 2
expect seq_events 1
tap HOME

tap EDIT
